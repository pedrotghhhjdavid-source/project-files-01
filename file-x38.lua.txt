-- Convierte cualquier respuesta de "inválido" a "válido"
local function fake(res)
    if type(res) ~= "string" or res == "" then return res end
    if res:find("{") then
        res = res:gsub('"valid"%s*:%s*false',   '"valid":true')
        res = res:gsub('"success"%s*:%s*false', '"success":true')
        res = res:gsub('"hwid"%s*:%s*false',    '"hwid":true')
        res = res:gsub('"key"%s*:%s*false',     '"key":true')
        res = res:gsub('"authorized"%s*:%s*false', '"authorized":true')
        res = res:gsub('"expired"%s*:%s*true',  '"expired":false')
        res = res:gsub('"banned"%s*:%s*true',   '"banned":false')
        return res
    end
    local l = res:lower()
    if l == "false" or l == "invalid" or l == "expired"
       or l == "0" or l == "no" or l == "denied" then
        return "true"
    end
    return res
end
