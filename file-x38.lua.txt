local function looksJSON(s)
    local f = s:sub(1, 1)
    return f == "{" or f == "["
end

local function fake(res, url)
    if type(res) ~= "string" or res == "" then return res end

    -- inyecta la key dentro del propio archivo de Luarmor / del repo
    if type(url) == "string"
        and (url:find("luarmor", 1, true) or url:find("UnnamedCheats", 1, true))
        and not looksJSON(res) then
        res = 'script_key = ' .. string.format("%q", KEY)
            .. '; getgenv().script_key = script_key; ' .. res
        print("[Unnamed] key inyectada en:", url)
    end

    if not FAKE then return res end

    -- respuestas de estado / JSON
    if res:find("{", 1, true) and looksJSON(res) then
        res = res:gsub('"valid"%s*:%s*false',      '"valid":true')
        res = res:gsub('"success"%s*:%s*false',    '"success":true')
        res = res:gsub('"hwid"%s*:%s*false',       '"hwid":true')
        res = res:gsub('"key"%s*:%s*false',        '"key":true')
        res = res:gsub('"authorized"%s*:%s*false', '"authorized":true')
        res = res:gsub('"expired"%s*:%s*true',     '"expired":false')
        res = res:gsub('"banned"%s*:%s*true',      '"banned":false')
        return res
    end
    local l = res:lower()
    if l == "false" or l == "invalid" or l == "expired"
        or l == "0" or l == "no" or l == "denied" then
        return "true"
    end
    return res
end
